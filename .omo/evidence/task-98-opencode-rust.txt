use oc_memory::{ENTRY_DELIMITER, MemoryStore, Operation, Scope, char_count, serialize};
use std::path::PathBuf;

/// Pad a realistic set of entries so the store lands on exactly `target` chars.
fn padded_to(target: usize, mut entries: Vec<String>) -> Vec<String> {
    let filler_prefix = "pad: ";
    let so_far = char_count(&serialize(&entries)) + ENTRY_DELIMITER.chars().count();
    let width = target - so_far - char_count(filler_prefix);
    entries.push(format!("{filler_prefix}{}", "-".repeat(width)));
    assert_eq!(char_count(&serialize(&entries)), target);
    entries
}

#[test]
fn print_rendered_blocks() {

    let mut global = MemoryStore::open(Scope::Global, dir.path().join("MEMORY.md")).expect("open");
